cmake_minimum_required(VERSION 3.21)
project(URL2KeyFill VERSION 3.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(U2K_BUILD_APP "Build native macOS application with real CEF" OFF)
option(U2K_DECKLINK "Compile real DeckLink SDK 16.0 backend" OFF)
add_library(u2k_core src/core/Core.cpp src/core/Pipeline.cpp src/core/Scene.cpp)
target_include_directories(u2k_core PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(u2k_core PUBLIC Threads::Threads)
enable_testing()
add_executable(core_tests tests/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE u2k_core)
add_test(NAME core COMMAND core_tests)
add_executable(scene_tests tests/SceneTests.cpp)
target_link_libraries(scene_tests PRIVATE u2k_core)
add_test(NAME scene COMMAND scene_tests)
if(U2K_BUILD_APP)
 if(NOT APPLE)
  message(FATAL_ERROR "Only the core is portable today; the Windows host is not implemented.")
 endif()
 enable_language(OBJCXX)
 set(PROJECT_ARCH arm64 CACHE STRING "CEF architecture")
 set(CEF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cef_binary_156.0.14+g4fa4ab5+chromium-156.0.8078.48_macosarm64_minimal" CACHE PATH "Pinned CEF distribution")
 if(NOT EXISTS "${CEF_ROOT}/include/cef_version.h")
  message(FATAL_ERROR "CEF missing: see docs/BUILD.md")
 endif()
 file(READ "${CEF_ROOT}/include/cef_version.h" CEF_VERSION_CONTENT)
 if(NOT CEF_VERSION_CONTENT MATCHES "156.0.14\\+g4fa4ab5\\+chromium-156.0.8078.48")
  message(FATAL_ERROR "Unexpected CEF version: use the pinned dependency.")
 endif()
 list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
 set(USE_SANDBOX ON CACHE BOOL "CEF sandbox must remain enabled" FORCE)
 find_package(CEF REQUIRED)
 add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
 add_library(u2k_browser src/browser/Browser.cpp)
 target_include_directories(u2k_browser PUBLIC "${CEF_ROOT}" src)
 target_link_libraries(u2k_browser PUBLIC u2k_core libcef_dll_wrapper)
 set(output_source src/output/Unavailable.cpp)
 if(U2K_DECKLINK)
  set(DECKLINK_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/DeckLinkSDK" CACHE PATH "Desktop Video 16.0 SDK root")
  find_path(DECKLINK_INCLUDE DeckLinkAPI.h PATHS "${DECKLINK_SDK_ROOT}/Mac/include" "${DECKLINK_SDK_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_file(DECKLINK_DISPATCH DeckLinkAPIDispatch.cpp PATHS "${DECKLINK_INCLUDE}" NO_DEFAULT_PATH REQUIRED)
  file(READ "${DECKLINK_INCLUDE}/DeckLinkAPIVersion.h" SDK_VERSION)
  if(NOT SDK_VERSION MATCHES "0x10000000")
   message(FATAL_ERROR "Expected Desktop Video SDK 16.0")
  endif()
  set(output_source src/output/DeckLink.cpp "${DECKLINK_DISPATCH}")
  add_executable(decklink_probe tests/Probe.cpp ${output_source})
  target_include_directories(decklink_probe PRIVATE "${DECKLINK_INCLUDE}")
  target_link_libraries(decklink_probe PRIVATE u2k_core "-framework CoreFoundation")
 endif()
 add_executable(URL2KeyFill MACOSX_BUNDLE src/platform/mac/Main.mm ${output_source})
 target_include_directories(URL2KeyFill PRIVATE "${CEF_ROOT}" src ${DECKLINK_INCLUDE})
 target_compile_options(URL2KeyFill PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
 target_link_libraries(URL2KeyFill PRIVATE u2k_browser ${CEF_STANDARD_LIBS} "-framework CoreFoundation" "-framework WebKit" "-framework QuartzCore")
 set_target_properties(URL2KeyFill PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
 set(CEF_APP "${CMAKE_CURRENT_BINARY_DIR}/URL2KeyFill.app")
 add_custom_command(TARGET URL2KeyFill POST_BUILD COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/package-framework.sh" "${CEF_BINARY_DIR}" "${CEF_APP}" VERBATIM)
 foreach(suffix IN LISTS CEF_HELPER_APP_SUFFIXES)
  string(REPLACE ":" ";" parts "${suffix}")
  list(GET parts 0 name)
  list(GET parts 1 target)
  list(GET parts 2 id)
  set(HELPER_NAME "URL2KeyFill Helper${name}")
  set(HELPER_ID "pt.lgsystems.url2keyfill.helper${id}")
  configure_file(cmake/Helper.plist.in "helper${target}.plist" @ONLY)
  add_executable(helper${target} MACOSX_BUNDLE src/platform/mac/Helper.mm)
  target_link_libraries(helper${target} PRIVATE u2k_browser ${CEF_STANDARD_LIBS})
  set_target_properties(helper${target} PROPERTIES OUTPUT_NAME "${HELPER_NAME}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/helper${target}.plist")
  add_dependencies(URL2KeyFill helper${target})
  add_custom_command(TARGET URL2KeyFill POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:helper${target}>" "${CEF_APP}/Contents/Frameworks/${HELPER_NAME}.app" VERBATIM)
 endforeach()
 set(U2K_LICENSE_FILES "${CEF_ROOT}/LICENSE.txt")
 if(U2K_DECKLINK)
  list(APPEND U2K_LICENSE_FILES "${DECKLINK_SDK_ROOT}/End User License Agreement.pdf")
 endif()
 file(GLOB_RECURSE U2K_ASSETS LIST_DIRECTORIES false CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*")
 # A changing manifest also detects removals; directory names ending in parentheses
 # must not become Make dependencies (Make interprets those as archive members).
 string(JOIN "\n" U2K_ASSET_MANIFEST ${U2K_ASSETS})
 file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/u2k-assets-manifest.txt" CONTENT "${U2K_ASSET_MANIFEST}\n" @ONLY)
 set_property(TARGET URL2KeyFill APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in" "${CMAKE_CURRENT_BINARY_DIR}/u2k-assets-manifest.txt" ${U2K_ASSETS})
 add_custom_command(TARGET URL2KeyFill POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CEF_APP}/Contents/Resources"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/URL2KeyFill.icns" "${CEF_APP}/Contents/Resources/URL2KeyFill.icns"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/assets/interface.html" "${CEF_APP}/Contents/Resources/interface.html"
  COMMAND ${CMAKE_COMMAND} -E rm -rf "${CEF_APP}/Contents/Resources/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CEF_APP}/Contents/Resources/assets"
  COMMAND ${CMAKE_COMMAND} -E copy ${U2K_LICENSE_FILES} "${CEF_APP}/Contents/Resources/"
  COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/sign-local.sh" "${CEF_APP}"
  VERBATIM)
endif()
